		icon = generic_mountain_fortification
		visible = {
			always = yes
		}
		
		available = {
			custom_trigger_tooltip = {
				tooltip =  ZBT_tt_out_of_space
				check_variable = { UC_unexcavated_sectors = 0 }
				check_variable = { UC_empty_sectors = 0 }
			}
			num_of_civilian_factories_available_for_projects > 2
		}

		cost = UC_new_layer_cost
		days_remove = UC_new_layer_time
		modifier = {
			civilian_factory_use = 3
		}
		remove_effect = {
			# set_variable = { var = UC_unexcavated_sectors value = 10 }
			add_to_variable = { var = UC_depth value = 1 }
			set_variable = { var = UC_empty_sectors value = 0 }
			IF = {
				limit = { check_variable = { UC_depth < 3 } }
				set_variable = { var = UC_unexcavated_sectors value = 5 }
			}
			ELSE_IF = {
				limit = { check_variable = { UC_depth < 5 } }
				set_variable = { var = UC_unexcavated_sectors value = 3 }
			}
			ELSE_IF = {
				limit = { check_variable = { UC_depth < 7 } }
				set_variable = { var = UC_unexcavated_sectors value = 2 }
			}
			ELSE = {
				set_variable = { var = UC_unexcavated_sectors value = 1 }
			}
			IF = {
				limit = {check_variable = { UC_depth < 7}}
				add_to_variable = {var = UC_unexcavated_sectors value = UC_modify_extra_sectors }
			}
			
			ZBT_UC_increase_costs = yes

			# Events to notify the player when they need to research heat tech
			IF = {
				limit = { check_variable = {UC_depth = 3} }
				country_event = ZBT_UU_events.1
			}
			IF = {
				limit = { check_variable = {UC_depth = 5} }
				country_event = ZBT_UU_events.2
			}
		}

			# Increment chance of finding special stuff
			IF = {
				limit = { check_variable = { STAT_excavations_so_far = 8 } }
				set_variable = { var = DW_discover_dragon_chance value = 10 }
			}
			ELSE_IF = {
				limit = { check_variable = { STAT_excavations_so_far > 8 } }
				add_to_variable = { var = DW_discover_dragon_chance value = 4 }
			}

			set_variable_to_random = {
				var = var:randomshit
				integer = yes
				min = 1
				max = 50
			}
			add_to_variable = { var = var:randomshit value = 1 }
			IF = {
				limit = { check_variable = { var:randomshit < var:STAT_resource_chance } }
				# TODO
			}
			IF = {
				limit = {
					check_variable = { var:randomshit < var:DW_discover_dragon_chance }
					NOT = { has_country_flag = DW_discovered }
				}
				set_country_flag = DW_discovered
				news_event = ZBT_DW_events.1
			}
		}
	}
