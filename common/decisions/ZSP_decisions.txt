ZSP_magic_cat = {

    ZSP_strengthen_industry = {

        icon = generic_construction

        ai_will_do = {
            base = 10
        }

        custom_cost_trigger = {
            check_variable = { ZSP_mana > 49 }
        }
        custom_cost_text = ZSP_use_50_mana

        available = {
            ZSP_has_available_spell_slot = yes
        }

        complete_effect = {
            add_to_variable = { ZSP_used_spell_slots = 1 }
            add_to_variable = { ZSP_mana = -50 }
        }

        days_remove = 70

        remove_effect = {
            random_owned_controlled_state = {
				add_extra_state_shared_building_slots = 1
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
			}
            add_to_variable = { ZSP_used_spell_slots = -1 }
        }

    }

    ZSP_strengthen_military_production = {

        icon = ger_military_buildup
        
        ai_will_do = {
            base = 10
        }

        custom_cost_trigger = {
            check_variable = { ZSP_mana > 49 }
        }
        custom_cost_text = ZSP_use_50_mana

        available = {
            ZSP_has_available_spell_slot = yes
        }

        complete_effect = {
            add_to_variable = { ZSP_used_spell_slots = 1 }
            add_to_variable = { ZSP_mana = -50 }
        }

        modifier = {
            industrial_capacity_factory = 0.25
        }

        days_remove = 30

        remove_effect = {
            add_to_variable = { ZSP_used_spell_slots = -1 }
        }

    }

    ZSP_burn_enemy_infrastructure = {

        icon = generic_ignite_civil_war

        ai_will_do = {
            base = 10
        }

        custom_cost_trigger = {
            check_variable = { ZSP_mana > 49 }
        }
        custom_cost_text = ZSP_use_50_mana

        available = {
            has_war = yes
            ZSP_has_available_spell_slot = yes
        }

        complete_effect = {
            add_to_variable = { ZSP_used_spell_slots = 1 }
            add_to_variable = { ZSP_mana = -50 }
            custom_effect_tooltip = ZSP_unlock_state_selector_tt
            set_country_flag = ZSP_burn_enemy_infrastructure_selector_flag
        }

        days_remove = 40

        remove_effect = {
            add_to_variable = { ZSP_used_spell_slots = -1 }
            clr_country_flag = ZSP_burn_enemy_infrastructure_selector_flag
        }

    }

    ZSP_burn_enemy_infrastructure_selector = {

        icon = infiltrate_state

        ai_will_do = {
            base = 10
        }

        visible = {
            has_country_flag = ZSP_burn_enemy_infrastructure_selector_flag
        }

        state_target = yes

        on_map_mode = map_only

        target_trigger = {
            FROM = {
                if = {
                    limit = {
                        any_country = {
                            owns_state = FROM
                        }
                    }
                    owner = {
                        controls_state = FROM
                        has_war_with = ROOT
                    }
                }
                else = {
                    always = no
                }
                non_damaged_building_level = {
                    building = infrastructure
                    level > 0
                }
            }
        }

        complete_effect = {
            clr_country_flag = ZSP_burn_enemy_infrastructure_selector_flag
            FROM = {
                damage_building = {
                    type = infrastructure
                    damage = 1
                }
                if = {
                    limit = {
                        non_damaged_building_level = {
                            building = industrial_complex
                            level > 0
                        }
                    }
                    damage_building = {
                        type = industrial_complex
                        damage = 1
                    }
                }
                if = {
                    limit = {
                        non_damaged_building_level = {
                            building = arms_factory
                            level > 0
                        }
                    }
                    damage_building = {
                        type = arms_factory
                        damage = 1
                    }
                }
            }
        }

    }

}
