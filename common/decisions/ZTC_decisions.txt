    ZTC_sell_guns_to_country = {

        icon = generic_prepare_civil_war

        custom_cost_trigger = {
            has_equipment = {
                infantry_equipment > 499
            }
        }
        custom_cost_text = decision_cost_guns_500

        available = {
            FROM = {
                has_war = yes
                NOT = { has_war_with = ROOT }
                has_opinion = {
                    target = ROOT
                    value > 9
                }
            }
            NOT = { has_country_flag = ZTC_currently_smuggling_guns }
        }

        targets_dynamic = yes

        target_trigger = {
            FROM = {
                has_war = yes
            }
        }

        days_re_enable = 200

        complete_effect = {
            set_country_flag = {
                flag = ZTC_currently_smuggling_guns
                days = 120
                value = 1
            }

            FROM = {
                country_event = { id = ZTC_events.5 days = 10 }
            }

            custom_effect_tooltip = if_they_accept_tt

            effect_tooltip = {
                send_equipment = {
                    equipment = infantry_equipment
                    amount = 500
                    target = FROM
                }
                add_cic = 15000
            }
        }
    }

}
