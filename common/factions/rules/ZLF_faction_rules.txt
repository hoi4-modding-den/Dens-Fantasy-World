ZLF_joining_rule_non_despotist = {
	type = joining_rules
	visible = {
		ZLF_is_despotist_ideology = no
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 0.1
			ZLF_is_despotist_ideology = no
		}
	}
	can_remove = {
		if = {
			limit = { original_tag = ZLF }
			ZLF_is_despotist_ideology = yes
		}
	}
	trigger = {
		ZLF_is_despotist_ideology = no
		if = {
			limit = { NOT = { has_rule = can_join_opposite_factions } }
			has_government = FROM
		}
	}
	modifier = {
		fascism_acceptance = -50
		neutrality_acceptance = -50
	}
}
