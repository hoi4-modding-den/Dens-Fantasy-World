on_actions = {

    on_civil_war_end = {
        effect = {
            if = {
                limit = { 
                    ROOT = { tag = ZLF } 
                    FROM = { tag = ZKN }
                }
                ROOT = { country_event = ZLF_cw_events.4 }
            }
        }
    }

    on_before_peace_conference_start = {
        effect = {
            if = {
                limit = {
                    FROM = { tag = ZLF }
                    ROOT = { NOT = { tag = ZKN } }
                }
                FROM = {
                    every_core_state = {
                        add_core_of = ZKN
                    }
                }
            }
            if = {
                limit = {
                    FROM = { tag = ZKN }
                    ROOT = { NOT = { tag = ZLF } }
                }
                FROM = {
                    every_core_state = {
                        add_core_of = ZLF
                    }
                }
            }
        }
    }

    on_capitulation = {
        effect = {
            if = {
                limit = {
                    ROOT = { tag = ZLF }
                    FROM = { has_idea = ZLF_hosting_exiled_royals }
                }
                ZKN = {
                    annex_country = {
                        target = ROOT
                        transfer_troops = no
                    }
                }
                FROM = {
                    remove_ideas = ZLF_hosting_exiled_royals
                    set_autonomy = {
                        target = ZKN
                        autonomous_state = autonomy_puppet
                        end_wars = no
                        end_civil_wars = no
                    }
                }
                ZKN = {
                    set_politics = {
                        ruling_party = neutrality
                        elections_allowed = no
                    }
                    load_focus_tree = ZLF_generic_focus_tree
                }
            }
        }
    }

    on_peaceconference_ended = {
        effect = {
            if = {
                limit = {
                    FROM = { tag = ZLF }
                }
                FROM = {
                    load_focus_tree = ZLF_generic_focus_tree
                    set_cosmetic_tag = ZLF_second_republic
                }
                random_other_country = {
                    limit = { has_character = ZLF_ekkehart_krippendorff }
                    ZLF_ekkehart_krippendorff = { set_nationality = FROM }
                }
                FROM = {
                    ZLF_ekkehart_krippendorff = {
                        add_country_leader_role = {
                            character = ZLF_ekkehart_krippendorff
                            promote_leader = yes
                            country_leader = {
                                ideology = moralism
                                expire = "1965.1.1.1"
                                traits = { ZLF_former_foreign_minister }
                            }
                        }
                    }
                    set_party_name = {
                        ideology = democratic
                        name = ZLF_democratic_party_NEP
                        long_name = ZLF_democratic_party_NEP_long
                    }
                }
            }
        }
    }

    on_war = {
        effect = {
            if = {
                limit = {
                    THIS = {
                        has_government = democratic
                        has_character = ZLF_michel_meisendorff
                        has_country_flag = ZLF_meisendorffs_defence_ministry_coup_unlocked
                    }
                }
                THIS = {
                    country_event = ZLF_events.17
                }
            }
        }
    }

    on_peace = {
        effect = {
            if = {
                limit = {
                    THIS = {
                        has_government = democratic
                        has_country_leader = { character = ZLF_michel_meisendorff ruling_only = yes }
                        has_country_flag = ZLF_trigger_election_after_war
                    }
                }
                THIS = {
                    country_event = { id = ZLF_events.18 days = 7 }
                }
            }
        }
    }

}
