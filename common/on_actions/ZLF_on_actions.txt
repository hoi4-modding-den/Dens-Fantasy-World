on_actions = {

    on_civil_war_end = {
        effect = {
            if = {
                limit = { 
                    ROOT = { tag = ZLF } 
                    FROM = { tag = ZKN }
                }
                ROOT = { country_event = ZLF_cw_events.4 }
            }
        }
    }

    on_capitulation = {
        effect = {
            if = {
                limit = {
                    ROOT = { tag = ZLF }
                    FROM = { has_idea = ZLF_hosting_exiled_royals }
                }
                ZKN = {
                    annex_country = {
                        target = ROOT
                        transfer_troops = no
                    }
                }
                FROM = {
                    remove_ideas = ZLF_hosting_exiled_royals
                    set_autonomy = {
                        target = ZKN
                        autonomous_state = autonomy_puppet
                        end_wars = no
                        end_civil_wars = no
                    }
                }
                ZKN = {
                    set_politics = {
                        ruling_party = neutrality
                        elections_allowed = no
                    }
                    load_focus_tree = generic_focus
                }
            }
        }
    }

}
