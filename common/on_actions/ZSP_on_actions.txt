on_actions = {

    on_weekly_ZSP = {
        effect = {
            if = {
                limit = {
                    check_variable = { modifier@ZSP_mana_generation > 0 }
                }
                add_to_variable = { ZSP_mana = modifier@ZSP_mana_generation }
            }
        }
    }

    on_monthly = {
        effect = {
            if = { # any country can use this one
                limit = {
                    check_variable = { modifier@ZSP_monthly_guns > 0 }
                }
                add_equipment_to_stockpile = { 
                    type = infantry_equipment_0
                    amount = modifier@ZSP_monthly_guns
                    producer = ZSP # so they are called muskets
                }
            }
        }
    }

    on_state_control_changed = {
        effect = {
            if = {
                limit = {
                    ROOT = { has_completed_focus = ZSP_world_domination }
                    FROM.FROM = {
                        any_neighbor_state = {
                            controller = {
                                NOT = { has_war_with = ROOT }
                                NOT = { is_ally_with = ROOT }
                            }
                        }
                    }
                }
                every_other_country = {
                    limit = {
                        is_neighbor_of = ROOT
                    }
                    ROOT = {
                        declare_war_on = {
                            target = PREV
                            type = ZSP_world_domination_goal
                        }
                    }
                }
            }
        }
    }

    on_peaceconference_ended = {
        effect = {
            if = {
                limit = {
                    FROM = { tag = ZSP }
                }
                if = {
                    limit = { has_cosmetic_tag = ZSP_shadow_realm }
                    drop_cosmetic_tag = yes
                }
                FROM = { load_focus_tree = ZLF_generic_focus_tree }
            }
        }
    }

    on_capitulation = {
        effect = {
            if = {
                limit = {
                    FROM = {
                        has_wargoal_against = {
                            target = ROOT
                            type = ZSP_world_domination_goal
                        }
                    }
                }
                if = {
                    limit = {
                        any_country = {
                            NOT = { tag = FROM }
                            has_war_with = ROOT
                        }
                    }
                    ROOT = {
                        every_owned_state = {
                            limit = { is_controlled_by = FROM }
                            transfer_state_to = FROM
                        }
                    }
                }
                else = {
                    FROM = {
                        annex_country = {
                            target = ROOT
                            transfer_troops = no
                        }
                    }
                }
            }
        }
    }

}
