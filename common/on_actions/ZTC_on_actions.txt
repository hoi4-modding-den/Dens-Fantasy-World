on_actions = {

    on_before_peace_conference_start = {
        effect = {
            if = {
                limit = {
                    ZTC = { has_completed_focus = ZTC_international_interventionism }
                    ROOT = {
                        has_volunteers_amount_from = {
                            tag = ZTC
                            count > 0
                        }
                    }
                }
                ZTC = {
                    country_event = ZTC_events.10
                }
            }
        }
    }

    on_capitulation = {
        effect = {
            if = {
                limit = {
                    ROOT = {
                        has_country_flag = ZTC_liberate_the_republic
                    }
                    FROM = {
                        original_tag = ZKN
                    }
                }
                FROM = {
                    every_core_state = {
                        add_core_of = ZLF
                    }
                }
                ROOT = {
                    annex_country = {
                        target = FROM
                        transfer_troops = no
                    }
                    release = ZLF
                    if = {
                        limit = { is_in_faction = yes }
                        add_to_faction = ZLF
                    }
                    else = {
                        create_faction_from_template = ZLF_faction_template_socialists
                        add_to_faction = ZLF
                    }
                }
            }
        }
    }

}
