### ENEMY ###

# This is where all enemy data is stored

ZAV_combat_init_enemy = {
    if = { 
        limit = {
            check_variable = { ZAV_combat_curr_enemy = 1 }
        }

        ZAV_combat_setup_enemy_1 = yes
    }
    else_if = { 
        limit = {
            check_variable = { ZAV_combat_curr_enemy = 2 }
        }

        ZAV_combat_setup_enemy_2 = yes
    }
}

ZAV_combat_setup_enemy_1 = {
    set_variable = { ZAV_combat_enemy_max_health = 37 }
    set_variable = { ZAV_combat_enemy_max_actions = 2 }

    clear_array = ZAV_combat_enemy_actions
    add_to_array = { ZAV_combat_enemy_actions = 1 }
    add_to_array = { ZAV_combat_enemy_actions = 2 }
}

ZAV_combat_setup_enemy_2 = {
    set_variable = { ZAV_combat_enemy_max_health = 58 }
    set_variable = { ZAV_combat_enemy_max_actions = 2 }

    clear_array = ZAV_combat_enemy_actions
    add_to_array = { ZAV_combat_enemy_actions = 3 }
    add_to_array = { ZAV_combat_enemy_actions = 4 }
    add_to_array = { ZAV_combat_enemy_actions = 5 }
}
