### HIT AND CRIT CHANCE ###

ZAV_combat_enemy_action_calculate_outcome = {
    multiply_temp_variable = { hit_chance = hit_chance_multiplier }
    
    round_temp_variable = hit_chance

    clamp_temp_variable = {
        var = hit_chance
        min = 0
        max = 100
    }

    set_temp_variable_to_random = {
        var = hit_roll
        min = 1
        max = 100
        integer = yes
    }

    if = {
        limit = { NOT = { check_variable = { hit_roll > hit_chance } } }
        
        set_temp_variable = { hit_result = 1 }
        
        set_temp_variable_to_random = {
            var = crit_roll
            min = 1
            max = 100
            integer = yes
        }

        if = {
            limit = { NOT = { check_variable = { crit_roll > crit_chance } } }
            set_temp_variable = { hit_result = 2 }
        }
    }

    set_temp_variable = { outcome = hit_result }
}

ZAV_combat_enemy_action_apply_critical = {
    if = {
        limit = { check_variable = { hit_result = 2 } }
        multiply_temp_variable = { damage = crit_damage_multiplier }
    }
}
