### ARTIFACT EFFECTS ###

# Static buffs such as +10 max health, these are things that are applied
# before the fight begins

# Max health - ZAV_combat_player_max_health
# Max wyrd - ZAV_combat_player_max_wyrd
# Max actions - ZAV_combat_player_max_actions
# Hit chance modifier - ZAV_player_hit_chance_modif
# Crit chance - ZAV_player_crit_chance
# Crit damage modifier - ZAV_player_crit_damage_modif

ZAV_combat_loadout_calculate_constant_buffs = {
    set_variable = { ZAV_combat_player_max_health = ZAV_player_max_health }
    set_variable = { ZAV_combat_player_max_wyrd = ZAV_player_max_wyrd }
    set_variable = { ZAV_combat_player_max_actions = ZAV_player_level }
    add_to_variable = { ZAV_combat_player_max_actions = 1 }
    set_variable = { ZAV_combat_player_hit_chance_modif = ZAV_player_hit_chance_modif }
    set_variable = { ZAV_combat_player_crit_chance = ZAV_player_crit_chance }
    set_variable = { ZAV_combat_player_crit_damage_modif = ZAV_player_crit_damage_modif }

    for_each_loop = {
        array = ZAV_combat_equipped_artifacts
        value = artifact

        if = {
            limit = { check_variable = { artifact = 11 } }
            add_to_variable = { ZAV_combat_player_max_wyrd = 10 }
        }
        else_if = {
            limit = { check_variable = { artifact = 7 } }
            add_to_variable = { ZAV_combat_player_max_health = 25 }
        }
        else_if = {
            limit = { check_variable = { artifact = 10 } }
            add_to_variable = { ZAV_combat_player_max_health = 6 }
        }
        else_if = {
            limit = { check_variable = { artifact = 5 } }
            add_to_variable = { ZAV_player_crit_chance = 10 }
        }
        else_if = {
            limit = { check_variable = { artifact = 12 } }
            add_to_variable = { ZAV_combat_player_max_health = 10 }
        }
        else_if = {
            limit = { check_variable = { artifact = 13 } }
            add_to_variable = { ZAV_combat_player_crit_chance = 10 }
            add_to_variable = { ZAV_combat_player_crit_damage_modif = 1 }
        }
        else_if = {
            limit = { check_variable = { artifact = 14 } }
            add_to_variable = { ZAV_combat_player_max_health = 15 }
            subtract_from_variable = { ZAV_combat_player_max_wyrd = 5 }
            add_to_variable = { ZAV_combat_player_max_actions = 1 }
        }
        else_if = {
            limit = { check_variable = { artifact = 15 } }
            add_to_variable = { ZAV_combat_player_max_wyrd = 10 }
        }
        else_if = {
            limit = { check_variable = { artifact = 16 } }
            add_to_variable = { ZAV_combat_player_crit_chance = 30 }
            add_to_variable = { ZAV_combat_player_crit_damage_modif = 1 }
        }
        else_if = {
            limit = { check_variable = { artifact = 18 } }
            add_to_variable = { ZAV_combat_player_max_wyrd = 10 }
        }
        else_if = {
            limit = { check_variable = { artifact = 19 } }
            add_to_variable = { ZAV_combat_player_hit_chance_modif = 0.4 }
        }
        else_if = {
            limit = { check_variable = { artifact = 21 } }
            add_to_variable = { ZAV_combat_player_max_health = 30 }
        }
        else_if = {
            limit = { check_variable = { artifact = 23 } }
            subtract_from_variable = { ZAV_combat_player_max_health = 20 }
            add_to_variable = { ZAV_combat_player_crit_chance = 50 }
        }
        else_if = {
            limit = { check_variable = { artifact = 25 } }
            add_to_variable = { ZAV_combat_player_crit_damage_modif = 3 }
        }
    }
}

# Static buffs such as +20% damage, these are factored into calculating
# the multiplier for each action

# Defense multiplier - ZAV_player_defense_multiplier (Note: this is not a temp variable)
# Damage multiplier - damage_multiplier
# Wyrd regen multiplier - wyrd_multiplier
# Health regen multiplier - healing_multiplier

ZAV_combat_loadout_calculate_multipliers = {
    for_each_loop = {
        array = ZAV_combat_equipped_artifacts
        value = artifact

        if = {
            limit = { check_variable = { artifact = 11 } }
            add_to_temp_variable = { healing_multiplier = 0.2 }
        }
        else_if = {
            limit = { check_variable = { artifact = 4 } }
            add_to_temp_variable = { damage_multiplier = 1.0 }
        }
        else_if = {
            limit = { check_variable = { artifact = 2 } }
            add_to_variable = { ZAV_player_defense_multiplier = 0.3 }
        }
        else_if = {
            limit = { check_variable = { artifact = 3 } }
            add_to_temp_variable = { wyrd_multiplier = 1.0 }
        }
        else_if = {
            limit = { check_variable = { artifact = 5 } }
            add_to_temp_variable = { wyrd_multiplier = 0.3 }
        }
        else_if = {
            limit = { check_variable = { artifact = 7 } }
            subtract_from_temp_variable = { healing_multiplier = 0.3 }
        }
        else_if = {
            limit = { check_variable = { artifact = 15 } }
            add_to_temp_variable = { wyrd_multiplier = 0.2 }
        }
        else_if = {
            limit = { check_variable = { artifact = 17 } }
            add_to_temp_variable = { healing_multiplier = 0.5 }
        }
        else_if = {
            limit = { check_variable = { artifact = 22 } }
            add_to_temp_variable = { damage_multiplier = 0.5 }
        }
        else_if = {
            limit = { check_variable = { artifact = 24 } }
            set_temp_variable = { half_health = ZAV_combat_player_max_health }
            divide_temp_variable = { half_health = 2 }
            round_temp_variable = half_health

            if = {
                limit = {
                    check_variable = { ZAV_combat_player_health < half_health }
                }
                add_to_temp_variable = { ZAV_player_defense_multiplier = 0.5 }
            }
        }
    }
}

# Constant effects that are applied on the player after each turn such
# as +2 health per turn

# health - health_change
# wyrd - wyrd_change

ZAV_combat_loadout_calculate_constants = {
    for_each_loop = {
        array = ZAV_combat_equipped_artifacts
        value = artifact

        if = {
            limit = { check_variable = { artifact = 4 } }
            subtract_from_temp_variable = { health_change = 5 }
        }
        else_if = {
            limit = { check_variable = { artifact = 3 } }
            set_temp_variable = { effect = 2 }
            set_temp_variable = { amount = 1 }
            ZAV_combat_player_get_status_effect = yes
        }
        else_if = {
            limit = { check_variable = { artifact = 12 } }
            random = {
                chance = 50
                set_temp_variable = { effect = 1 }
                set_temp_variable = { amount = 1 }
                ZAV_combat_inflict_status_effect_on_enemy = yes
            }
        }
        else_if = {
            limit = { check_variable = { artifact = 13 } }
            subtract_from_temp_variable = { wyrd_change = 2 }
        }
        else_if = {
            limit = { check_variable = { artifact = 19 } }
            subtract_from_temp_variable = { wyrd_change = 6 }
        }
        else_if = {
            limit = { check_variable = { artifact = 20 } }
            random = {
                chance = 50
                set_temp_variable = { effect = 5 }
                set_temp_variable = { amount = 1 }
                ZAV_combat_inflict_status_effect_on_enemy = yes
            }
        }
        else_if = {
            limit = { check_variable = { artifact = 23 } }
            set_temp_variable = { effect = 2 }
            set_temp_variable = { amount = 1 }
            ZAV_combat_player_get_status_effect = yes
        }
    }
}

### UTILITY ###

ZAV_combat_loadout_init = {
    clear_array = ZAV_combat_owned_artifacts
    clear_array = ZAV_combat_equipped_artifacts

    for_each_loop = {
        array = ZAV_vault_artifacts
        value = artifact

        add_to_array = { ZAV_combat_owned_artifacts = artifact }
    }

    set_variable = { ZAV_combat_max_equipped_artifacts = 2 }
    multiply_variable = { ZAV_combat_max_equipped_artifacts = ZAV_player_level }
}
