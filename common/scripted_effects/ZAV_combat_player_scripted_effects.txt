# This is where all player actions are defined

ZAV_combat_do_player_actions = {
    set_temp_variable = { player = 0 }

    if = {
        limit = {
            is_in_array = { ZAV_player_selected_actions = 1 }
        }

        set_temp_variable = { action = 1 }
        set_temp_variable = { hit_result = 0 }
        set_temp_variable = { hit_chance = 80 }

        ZAV_combat_action_calculate_outcome = yes

        if = {
            limit = { check_variable = { hit_result > 0 } }

            set_temp_variable = { damage = 5 }

            ZAV_combat_action_apply_critical = yes

            multiply_temp_variable = { damage = damage_multiplier }
            divide_temp_variable = { damage = ZAV_enemy_defense_multiplier }
            round_temp_variable = damage

            add_to_temp_variable = { total_damage = damage }

            set_temp_variable = { stat = damage }
            ZAV_combat_log_action = yes

            ZAV_combat_attack = yes
        }
        else = {
            set_temp_variable = { stat = 0 }
            ZAV_combat_log_action = yes
        }
    }
    if = {
        limit = {
            is_in_array = { ZAV_player_selected_actions = 2 }
        }

        set_temp_variable = { action = 2 }

        set_temp_variable = { wyrd_regen = 4 }

            is_in_array = { ZAV_player_selected_actions = 4 }

        set_temp_variable = { action = 4 }
        set_temp_variable = { hit_result = 0 }
        set_temp_variable = { hit_chance = 95 }

                chance = 20

            }
        }

            set_temp_variable = { stat = 0 }

### HIT AND CRIT CHANCE ###

ZAV_combat_action_calculate_outcome = {
    multiply_temp_variable = { hit_chance = ZAV_player_hit_chance_modif }
    
    round_temp_variable = hit_chance

    clamp_temp_variable = {
        var = hit_chance
        min = 0
        max = 100
    }

    set_temp_variable_to_random = {
        var = hit_roll
        min = 1
        max = 100
        integer = yes
    }

    if = {
        limit = { NOT = { check_variable = { hit_roll > hit_chance } } }
        
        set_temp_variable = { hit_result = 1 }
        
        set_temp_variable_to_random = {
            var = crit_roll
            min = 1
            max = 100
            integer = yes
        }

        if = {
            limit = { NOT = { check_variable = { crit_roll > ZAV_combat_player_crit_chance } } }
            set_temp_variable = { hit_result = 2 }
        }
    }

    set_temp_variable = { outcome = hit_result }
}

ZAV_combat_action_apply_critical = {
    if = {
        limit = { check_variable = { hit_result = 2 } }
        multiply_temp_variable = { damage = ZAV_combat_player_crit_damage_modif }
    }
}

### DATA ###

# Calculates the set of actions the player can do at the start of the combat
ZAV_combat_calculate_player_actions = {
    clear_array = ZAV_combat_available_actions
    
    # Basic actions
    add_to_array = { ZAV_combat_available_actions = 1 }  # PISTOL
    add_to_array = { ZAV_combat_available_actions = 2 }  # PRAY
    add_to_array = { ZAV_combat_available_actions = 3 }  # HEAL
    add_to_array = { ZAV_combat_available_actions = 4 }  # BARRAGE

    # Actions enabled by artifacts and blessings
    if = {
        limit = {
            is_in_array = { ZAV_combat_equipped_artifacts = 3 }
        }
        add_to_array = { ZAV_combat_available_actions = 5 }  # CORRUPT
    }
    
    if = {
        limit = {
            is_in_array = { ZAV_combat_equipped_artifacts = 11 }
        }
        add_to_array = { ZAV_combat_available_actions = 6 }  # JUSTICE
    }
}
