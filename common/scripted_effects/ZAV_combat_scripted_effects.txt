### SETUP ###

# These are the stats the player starts with at the start of the game
ZAV_combat_setup_starting_stats = {
    set_variable = { ZAV_player_max_health = 28 }
    set_variable = { ZAV_player_max_wyrd = 15 }
    set_variable = { ZAV_player_max_actions = 2 }
    set_variable = { ZAV_player_hit_chance_modif = 1 }
    set_variable = { ZAV_player_crit_chance = 3 }
    set_variable = { ZAV_player_crit_damage_modif = 2 }

    ZAV_combat_setup_actions_data = yes
}

# This initialises the window (starting with loadout)
ZAV_combat_init = {
    if = {
        limit = {
            is_ai = no
        }
        clr_country_flag = ZAV_combat_victory
        clr_country_flag = ZAV_combat_defeat
        set_country_flag = ZAV_combat_ongoing
        clr_country_flag = ZAV_combat_loadout_complete
        clr_country_flag = ZAV_combat_enemy_turn

        clear_array = ZAV_player_selected_actions
        
        ZAV_combat_loadout_init = yes
    }
    else = {
        set_country_flag = ZAV_combat_victory
        ZAV_combat_finished_effect = yes
    }
}

# This starts the combat after choosing a loadout
ZAV_combat_start = {
    clear_array = ZAV_combat_log_players
    clear_array = ZAV_combat_log_actions
    clear_array = ZAV_combat_log_outcomes
    clear_array = ZAV_combat_log_stats

    clear_array = ZAV_combat_player_status_effects
    clear_array = ZAV_combat_player_status_effects_amounts
    clear_array = ZAV_combat_enemy_status_effects
    clear_array = ZAV_combat_enemy_status_effects_amounts

    ZAV_combat_loadout_calculate_constant_buffs = yes

    set_variable = { ZAV_combat_player_health = ZAV_combat_player_max_health }
    set_variable = { ZAV_combat_player_wyrd = ZAV_combat_player_max_wyrd }
    set_variable = { ZAV_combat_remaining_wyrd = ZAV_combat_player_wyrd }
    set_variable = { ZAV_player_defense_multiplier = 1 }
    set_variable = { ZAV_combat_remaining_actions = ZAV_combat_player_max_actions }

    ZAV_combat_init_enemy = yes

    set_variable = { ZAV_combat_enemy_health = ZAV_combat_enemy_max_health }
    set_variable = { ZAV_enemy_defense_multiplier = 1 }

    ZAV_combat_calculate_player_actions = yes
    ZAV_combat_update_stats = yes

    set_country_flag = ZAV_combat_loadout_complete
}

### UTILITY ###

# Updates the stats area of both the player and the enemy
ZAV_combat_update_stats = {
    set_temp_variable = { player_health_frame = 100 }
    set_temp_variable = { player_wyrd_frame = 100 }
    set_temp_variable = { enemy_health_frame = 100 }

    divide_temp_variable = { player_health_frame = ZAV_combat_player_max_health }
    divide_temp_variable = { player_wyrd_frame = ZAV_combat_player_max_wyrd }
    divide_temp_variable = { enemy_health_frame = ZAV_combat_enemy_max_health }

    multiply_temp_variable = { player_health_frame = ZAV_combat_player_health }
    multiply_temp_variable = { player_wyrd_frame = ZAV_combat_player_wyrd }
    multiply_temp_variable = { enemy_health_frame = ZAV_combat_enemy_health }

    round_temp_variable = player_health_frame
    round_temp_variable = player_wyrd_frame
    round_temp_variable = enemy_health_frame

    set_variable = { ZAV_player_health_frame = player_health_frame }
    set_variable = { ZAV_player_wyrd_frame = player_wyrd_frame }
    set_variable = { ZAV_enemy_health_frame = enemy_health_frame }
}

# Logs an action in the combat log
ZAV_combat_log_action = {
    add_to_array = { ZAV_combat_log_players = player }
    add_to_array = { ZAV_combat_log_actions = action }
    add_to_array = { ZAV_combat_log_outcomes = outcome }
    add_to_array = { ZAV_combat_log_stats = stat }
}

### MULTIPLIERS AND CONSTANTS ###

ZAV_combat_calculate_player_multipliers = {
    set_variable = { ZAV_player_defense_multiplier = 1 }
    set_temp_variable = { damage_multiplier = 1 }
    set_temp_variable = { healing_multiplier = 1 }
    set_temp_variable = { wyrd_multiplier = 1 }
    set_temp_variable = { hit_chance_multiplier = ZAV_combat_player_hit_chance_modif }
    set_temp_variable = { crit_chance = ZAV_combat_player_crit_chance }
    set_temp_variable = { crit_damage_multiplier = ZAV_combat_player_crit_damage_modif }

    ZAV_combat_loadout_calculate_multipliers = yes
    ZAV_combat_status_effects_calculate_multipliers = yes

    if = {
        limit = { has_country_flag = ZAV_debug_log }
        log = "Player multipliers"
        log = "defense_multiplier: [?ZAV_player_defense_multiplier]"
        log = "damage_multiplier: [?damage_multiplier]"
        log = "healing_multiplier: [?healing_multiplier]"
        log = "wyrd_multiplier: [?wyrd_multiplier]"
        log = "hit_chance_multiplier: [?hit_chance_multiplier]"
        log = "crit_chance: [?crit_chance]"
        log = "crit_damage_multiplier: [?crit_damage_multiplier]"
    }
}

### PLAYER TURN ###

ZAV_combat_do_player_turn = {
    set_temp_variable = { total_damage = 0 }
    set_temp_variable = { total_wyrd_cost = 0 }
    set_temp_variable = { total_wyrd_regen = 0 }
    set_temp_variable = { total_health_regen = 0 }

    # Calculate effects from status effects and artifacts
    ZAV_combat_calculate_player_multipliers = yes

    # Do actions that the player selected
    ZAV_combat_do_player_actions = yes

    log = "total_wyrd_cost: [?total_wyrd_cost]"
    log = "total_wyrd_regen: [?total_wyrd_regen]"
    log = "total_health_regen: [?total_health_regen]"
    log = "total_damage: [?total_damage]"

    # Apply the effects from the actions
    subtract_from_variable = { ZAV_combat_player_wyrd = total_wyrd_cost }
    add_to_variable = { ZAV_combat_player_wyrd = total_wyrd_regen }
    add_to_variable = { ZAV_combat_player_health = total_health_regen }
    subtract_from_variable = { ZAV_combat_enemy_health = total_damage }

    # Calculate and apply constants from effects and artifacts
    ZAV_combat_calculate_player_constants = yes

    # Reduce all status effects by 1
    ZAV_combat_reduce_player_status_effect_amounts = yes

    # Recalculate multipliers for next turn
    ZAV_combat_calculate_player_multipliers = yes

    # Clamp variables to ensure they're within range
    clamp_variable = {
        var = ZAV_combat_player_health
        min = 0
        max = ZAV_combat_player_max_health
    }

    clamp_variable = {
        var = ZAV_combat_player_wyrd
        min = 0
        max = ZAV_combat_player_max_wyrd
    }

    clamp_variable = {
        var = ZAV_combat_enemy_health
        min = 0
        max = ZAV_combat_enemy_max_health
    }

    # Reset actions and resources for next turn
    clear_array = ZAV_player_selected_actions
    set_variable = { ZAV_combat_remaining_wyrd = ZAV_combat_player_wyrd }
    set_variable = { ZAV_combat_remaining_actions = ZAV_combat_player_max_actions }

    # Update stats and look for win conditions
    ZAV_combat_update_stats = yes
    ZAV_combat_check_for_win = yes
}

### ENEMY TURN ###

ZAV_combat_do_enemy_turn = {
    set_temp_variable = { total_damage = 0 }
    set_temp_variable = { total_health_regen = 0 }

    # Calculate multipliers from status effects
    ZAV_combat_calculate_enemy_multipliers = yes

    if = {
        limit = { has_country_flag = ZAV_debug_log }
        log = "Enemy Actions"
    }

    # Choose max actions amount of random actions from their list
    for_loop_effect = {
        end = ZAV_combat_enemy_max_actions
        value = action
        set_temp_variable_to_random = {
            var = enemy_action
            min = 0
            max = ZAV_combat_enemy_actions^num
            integer = yes
        }

        set_temp_variable = { enemy_action = ZAV_combat_enemy_actions^enemy_action }

        if = {
            limit = { has_country_flag = ZAV_debug_log }
            log = "action id = [?enemy_action]"
        }

        ZAV_combat_do_enemy_action = yes
    }

    add_to_variable = { ZAV_combat_enemy_health = total_health_regen }
    subtract_from_variable = { ZAV_combat_player_health = total_damage }

    # Calculate and apply constants from status effects
    ZAV_combat_calculate_enemy_constants = yes

    # Reduce all status effects by 1
    ZAV_combat_reduce_enemy_status_effect_amounts = yes

    # Recalculate multipliers for next turn
    ZAV_combat_calculate_enemy_multipliers = yes

    # Clamp variables to ensure they're within range
    clamp_variable = {
        var = ZAV_combat_enemy_health
        min = 0
        max = ZAV_combat_enemy_max_health
    }

    clamp_variable = {
        var = ZAV_combat_player_health
        min = 0
        max = ZAV_combat_player_max_health
    }

    # Update stats and look for win condition
    ZAV_combat_update_stats = yes
    ZAV_combat_check_for_win = yes
}

### WIN/LOSE STATES ###

ZAV_combat_check_for_win = {
    if = {
        limit = {
            check_variable = { ZAV_combat_enemy_health < 1 }
        }

        set_country_flag = ZAV_combat_victory
    }
    else_if = {
        limit = {
            check_variable = { ZAV_combat_player_health < 1 }
        }

        set_country_flag = ZAV_combat_defeat
    }
}
