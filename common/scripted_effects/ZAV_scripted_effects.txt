### NOTEBOOK ###

ZAV_add_new_divine_saviours_info = {
    if = {
        limit = {
            NOT = {
                has_country_flag = ZAV_open_notebook
            }
        }

        set_country_flag = ZAV_new_notebook_discovery
    }

    if = {
        limit = {
            NOT = {
                check_variable = { ZAV_notebook_tab = 0 }
            }
        }

        set_country_flag = ZAV_new_divine_saviours_discovery
    }
}
