### ARTIFACTS ###

ZAV_add_artifact_to_vault = {
    if = {
        limit = {
            NOT = {
                is_in_array = { ZAV_vault_artifacts = ZAV_active_artifact }
            }
        }

        custom_effect_tooltip = ZAV_add_artifact_to_vault_tt
        add_to_array = { ZAV_vault_artifacts = ZAV_active_artifact }
    }
    else = {
        # Fallback (shouldn't happen)
        custom_effect_tooltip = ZAV_cant_add_artifact_to_vault_tt
    }
}

ZAV_remove_artifact_from_vault = {
    custom_effect_tooltip = ZAV_remove_artifact_from_vault_tt
    
    if = {
        limit = {
            is_in_array = { ZAV_vault_artifacts = ZAV_active_artifact }
        }

        remove_from_array = { ZAV_vault_artifacts = ZAV_active_artifact }
    }
    #else = {
    #    custom_effect_tooltip = ZAV_cant_remove_artifact_from_vault_tt
    #}
}

### NOTEBOOK ###

ZAV_add_new_divine_saviours_info = {
    custom_effect_tooltip = ZAV_notebook_updated_tt

    if = {
        limit = {
            NOT = {
                has_country_flag = ZAV_open_notebook
            }
        }

        set_country_flag = ZAV_new_notebook_discovery
    }

    if = {
        limit = {
            NOT = {
                check_variable = { ZAV_notebook_tab = 0 }
            }
        }

        set_country_flag = ZAV_new_divine_saviours_discovery
    }
}
