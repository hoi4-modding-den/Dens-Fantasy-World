ZBT_UC_apply_buffs_civ = {
	# tempvar setup
	set_temp_variable = { temp_construction_speed = 0 }
	# calculate cumulative buff
	IF = {
		limit = { 
			OR = {
				has_country_flag = UU_improved_civs1
				check_variable = { temp_var:fake_buff_for_tt = 1 }
			}
		}
		add_to_temp_variable = {var = temp_construction_speed value = 0.01}
		# add_to_temp_variable = {var = temp_construction_speed value = temp_var:fake_buff_for_tt }
	}

	# apply buff
	add_to_variable = {var = UC_construction_speed value = temp_var:temp_construction_speed tooltip = production_speed_buildings_factor_tt}
}

ZBT_UC_recalc_costs = {
	# adds the flat time/cost increases to the base time/cost
	set_variable = { var = UC_new_layer_cost value = UC_base_new_layer_cost }
	add_to_variable = { var = UC_new_layer_cost value = UC_modify_cost }
	
	set_variable = { var = UC_new_layer_time value = UC_base_new_layer_time }
	add_to_variable = { var = UC_new_layer_time value = UC_modify_time }
	
	set_variable = { var = UC_sector_dig_cost value = UC_base_sector_dig_cost }
	add_to_variable = { var = UC_sector_dig_cost value = UC_modify_cost }
	
	set_variable = { var = UC_sector_dig_time value = UC_base_sector_dig_time }
	add_to_variable = { var = UC_sector_dig_time value = UC_modify_sector_dig_time }
	
	set_variable = { var = UC_sector_build_cost value = UC_base_sector_build_cost }
	add_to_variable = { var = UC_sector_build_cost value = UC_modify_cost }
	
	set_variable = { var = UC_sector_build_time value = UC_base_sector_build_time }
	add_to_variable = { var = UC_sector_build_time value = UC_modify_time }
	
}

ZBT_UC_increase_costs = {
	multiply_variable = { var = UC_base_new_layer_cost value = 1.1 }
	round_variable = UC_new_layer_cost
	multiply_variable = { var = UC_base_new_layer_time value = 1.05 }
	round_variable = UC_new_layer_time
	multiply_variable = { var = UC_base_sector_dig_cost value = 1.1 }
	round_variable = UC_sector_dig_cost
	multiply_variable = { var = UC_base_sector_dig_time value = 1.05 }
	round_variable = UC_sector_dig_time
	multiply_variable = { var = UC_base_sector_build_cost value = 1.1 }
	round_variable = UC_sector_build_cost
	multiply_variable = { var = UC_base_sector_build_time value = 1.05 }
	round_variable = UC_sector_build_time
	ZBT_UC_recalc_costs = yes
}
