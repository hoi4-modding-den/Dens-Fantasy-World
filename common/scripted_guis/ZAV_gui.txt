    ZAV_combat_screen = {
        context_type = player_context
        window_name = "ZAV_combat_window"

        visible = {
            original_tag = ZAV
            has_country_flag = ZAV_combat_ongoing
            has_country_flag = ZAV_combat_loadout_complete
        }

        dynamic_lists = {
            ZAV_combat_actions_gridbox = {
                array = ZAV_combat_available_actions
                entry_container = "ZAV_combat_action_entry"
                index = entry_index
                value = entry_value
            }

            ZAV_combat_log_gridbox = {
                array = ZAV_combat_log_actions
                entry_container = "ZAV_combat_log_entry"
                index = entry_index
                value = entry_value
            }

            ZAV_combat_player_status_effects_gridbox = {
                array = ZAV_combat_player_status_effects
                entry_container = "ZAV_combat_status_effect_entry"
                index = entry_index
                value = entry_value
            }

            ZAV_combat_enemy_status_effects_gridbox = {
                array = ZAV_combat_enemy_status_effects
                entry_container = "ZAV_combat_status_effect_enemy_entry"
                index = entry_index
                value = entry_value
            }
        }

        triggers = {
            ZAV_combat_action_button_click_enabled = {
                OR = {
                    AND = {
                        check_variable = {
                            var = ZAV_combat_remaining_wyrd
                            value = ZAV_combat_actions_wyrd_costs^entry_value
                            compare = greater_than_or_equals
                        }
                        check_variable = { ZAV_combat_remaining_actions > 0 }
                    }
                    is_in_array = { ZAV_player_selected_actions = entry_value }
                }
            }
        }

        properties = {
            ZAV_combat_enemy_image = {
                image = "[GetCombatEnemyImage]"
            }

            ZAV_combat_status_effect_icon = {
                image = "[GetCombatStatusEffectImage]"
            }

            ZAV_combat_status_effect_enemy_icon = {
                image = "[GetCombatStatusEffectImage]"
            }

            ZAV_combat_player_health_bar = {
                frame = ZAV_player_health_frame
            }

            ZAV_combat_player_wyrd_bar = {
                frame = ZAV_player_wyrd_frame
            }

            ZAV_combat_enemy_health_bar = {
                frame = ZAV_enemy_health_frame
            }

            ZAV_combat_action_button = {
                image = "[GetCombatActionBackground]"
            }

            ZAV_combat_action_entry_icon = {
                image = "[GetCombatActionIcon]"
            }
        }

        effects = {
            ZAV_combat_end_turn_button_click = {
                if = {
                    limit = {
                        OR = {
                            has_country_flag = ZAV_combat_victory
                            has_country_flag = ZAV_combat_defeat
                        }
                    }
                    clr_country_flag = ZAV_combat_ongoing
                    ZAV_combat_finished_effect = yes
                }
                else_if = {
                    limit = { has_country_flag = ZAV_combat_enemy_turn }
                    ZAV_combat_do_enemy_turn = yes
                    clr_country_flag = ZAV_combat_enemy_turn
                }
                else = {
                    ZAV_combat_do_player_turn = yes
                    set_country_flag = ZAV_combat_enemy_turn
                }
            }

            ZAV_combat_action_button_click = {
                sound_effect = "click_default"
                
                if = {
                    limit = {
                        is_in_array = { ZAV_player_selected_actions = entry_value }
                    }

                    remove_from_array = { ZAV_player_selected_actions = entry_value } 
                    add_to_variable = { ZAV_combat_remaining_wyrd = ZAV_combat_actions_wyrd_costs^entry_value }
                    add_to_variable = { ZAV_combat_remaining_actions = 1 }
                }
                else = {
                    add_to_array = { ZAV_player_selected_actions = entry_value }
                    subtract_from_variable = { ZAV_combat_remaining_wyrd = ZAV_combat_actions_wyrd_costs^entry_value }
                    subtract_from_variable = { ZAV_combat_remaining_actions = 1 }
                }
            }
        }
    }

    ZAV_combat_loadout_screen = {
        context_type = player_context
        window_name = "ZAV_combat_loadout_window"

        visible = {
            original_tag = ZAV
            has_country_flag = ZAV_combat_ongoing
            NOT = { has_country_flag = ZAV_combat_loadout_complete }
        }

        dynamic_lists = {
            ZAV_combat_loadout_owned_artifacts_gridbox = {
                array = ZAV_combat_owned_artifacts
                entry_container = "ZAV_combat_loadout_artifact_entry"
                index = artifact_index
                value = artifact_value
            }

            ZAV_combat_loadout_equipped_artifacts_gridbox = {
                array = ZAV_combat_equipped_artifacts
                entry_container = "ZAV_combat_loadout_artifact_entry"
                index = artifact_index
                value = artifact_value
            }
        }

        effects = {
            ZAV_combat_loadout_start_button_click = {
                ZAV_combat_start = yes
            }

            ZAV_combat_loadout_artifact_icon_click = {
                if = {
                    limit = {
                        is_in_array = { ZAV_combat_owned_artifacts = artifact_value }
                        check_variable = { ZAV_combat_equipped_artifacts^num < ZAV_combat_max_equipped_artifacts }
                    }

                    remove_from_array = { ZAV_combat_owned_artifacts = artifact_value }
                    add_to_array = { ZAV_combat_equipped_artifacts = artifact_value }
                }
                else_if = {
                    limit = {
                        is_in_array = { ZAV_combat_equipped_artifacts = artifact_value }
                    }

                    remove_from_array = { ZAV_combat_equipped_artifacts = artifact_value }
                    add_to_array = { ZAV_combat_owned_artifacts = artifact_value }
                }
            }
        }

        triggers = {
            ZAV_combat_loadout_artifact_icon_click_enabled = {
                set_temp_variable = { artifact_power = 6 }
                subtract_from_temp_variable = { artifact_power = artifact_classes^artifact_value }
                NOT = { check_variable = { artifact_power > ZAV_player_level } }
            }
        }

        properties = {
			ZAV_combat_loadout_artifact_icon = {
                image = "[GetVaultArtifactIcon]"
            }
		}
    }
